cmake_minimum_required(VERSION 3.16)
project(MirvkBuntuDependencyScout VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(mirvkbuntu-scout main.cpp)
if(WIN32)
  target_compile_definitions(mirvkbuntu-scout PRIVATE MIRVKBUNTU_SCOUT_WINDOWS=1)
elseif(APPLE)
  target_compile_definitions(mirvkbuntu-scout PRIVATE MIRVKBUNTU_SCOUT_MACOS=1)
else()
  target_compile_definitions(mirvkbuntu-scout PRIVATE MIRVKBUNTU_SCOUT_LINUX=1)
endif()
include(GNUInstallDirs)
install(TARGETS mirvkbuntu-scout RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
